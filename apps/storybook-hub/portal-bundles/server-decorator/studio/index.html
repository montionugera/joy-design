Test downstream fault tolerance. Trigger successes and failures to watch the circuit breaker transition through CLOSED, OPEN, and HALF-OPEN states with automatic cooldowns.

Polyglot Code Implementation

// TypeScript / Node
import { withCircuitBreaker, CircuitBreakerOpenError } from '@montionugera/server-decorator';

const resilientFetch = withCircuitBreaker(async (id: string) => {
  return await fetchUserFromRemote(id);
}, {
  threshold: 3,
  resetTimeoutMs: 3000,
  fallback: () => ({ id: 'fallback', name: 'Cached User' })
});

Polyglot Code Implementation

// TypeScript / Node
import { withBatchLoad } from '@montionugera/server-decorator';

// Microservice callers use standard single-fetch signature:
export const getUserById = withBatchLoad(async (ids: string[]) => {
  console.log('Fired single batched database query for:', ids);
  return await db.query('SELECT * FROM users WHERE id IN (?)', [ids]);
});

// Elsewhere in parallel resolvers:
const [u1, u2, u3] = await Promise.all([
  getUserById('u_1'),
  getUserById('u_2'),
  getUserById('u_1') // Deduped & coalesced!
]);
